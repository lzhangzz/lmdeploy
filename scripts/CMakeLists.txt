cmake_minimum_required(VERSION 3.25.2)
project(DSLRMSNormVerification LANGUAGES C CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "These dynamic-loading verification applications currently support Linux only")
endif()

find_package(CUDAToolkit REQUIRED)
include(CTest)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
get_filename_component(REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}" DIRECTORY)

option(ENABLE_CUTE_DSL "Build the CuTe DSL RMSNorm verification application" ON)
option(ENABLE_TILELANG "Build the TileLang RMSNorm verification application" ON)
option(ENABLE_TRITON "Build the Triton RMSNorm verification application" ON)

set(CUTE_DSL_ROOT "" CACHE PATH "CuTe DSL CUDA variant directory containing include/ and lib/")
set(CUTE_DSL_CUDA_ROOT "" CACHE PATH "Optional matching CUDA headers/runtime root for CuTe DSL")
set(TVM_FFI_ROOT "" CACHE PATH "TVM FFI package directory containing include/ and lib/")
set(TILELANG_ROOT "" CACHE PATH "TileLang package directory containing lib/libtvm.so")
set(CUTE_DSL_ARTIFACT_DIR "${REPOSITORY_ROOT}/build/cute_dsl_rmsnorm" CACHE PATH "Exported CuTe DSL kernels")
set(TILELANG_ARTIFACT_DIR "${REPOSITORY_ROOT}/build/tilelang_rmsnorm" CACHE PATH "Exported TileLang kernels")
set(TRITON_ARTIFACT_DIR "${REPOSITORY_ROOT}/build/triton_rmsnorm" CACHE PATH "Exported Triton kernels")

function(require_artifact path exporter)
    if(NOT EXISTS "${path}")
        message(FATAL_ERROR "Missing ${path}. Export kernels with ${exporter} before configuring CMake")
    endif()
endfunction()

function(add_rmsnorm_application target)
    add_executable(${target} "${target}.cpp")
    target_compile_features(${target} PRIVATE cxx_std_17)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    target_compile_options(${target} PRIVATE -Wall -Wextra)
    target_link_libraries(${target} PRIVATE CUDA::cuda_driver ${CMAKE_DL_LIBS})
endfunction()

function(configure_rmsnorm_test test)
    set(modifications "PYTHONPATH=unset:")
    foreach(directory IN LISTS ARGN)
        list(APPEND modifications "LD_LIBRARY_PATH=path_list_prepend:${directory}")
    endforeach()
    set_tests_properties(${test} PROPERTIES
        RUN_SERIAL TRUE
        ENVIRONMENT_MODIFICATION "${modifications}")
endfunction()

if(ENABLE_CUTE_DSL)
    if(NOT CUTE_DSL_ROOT)
        message(FATAL_ERROR "Set CUTE_DSL_ROOT to the CuTe DSL CUDA variant directory, or disable ENABLE_CUTE_DSL")
    endif()
    find_path(CUTE_DSL_INCLUDE_DIR CuteDSLRuntime.h PATHS "${CUTE_DSL_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
    find_library(CUTE_DSL_RUNTIME_LIBRARY cute_dsl_runtime PATHS "${CUTE_DSL_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    get_filename_component(CUTE_DSL_LIBRARY_DIR "${CUTE_DSL_RUNTIME_LIBRARY}" DIRECTORY)
    add_library(CuteDSLRuntime SHARED IMPORTED)
    set_target_properties(CuteDSLRuntime PROPERTIES
        IMPORTED_LOCATION "${CUTE_DSL_RUNTIME_LIBRARY}"
        INTERFACE_SYSTEM_INCLUDE_DIRECTORIES "${CUTE_DSL_INCLUDE_DIR}"
        INTERFACE_INCLUDE_DIRECTORIES "${CUTE_DSL_INCLUDE_DIR}")

    set(CUTE_CUDA_TARGET CUDA::cudart)
    set(CUTE_CUDA_INCLUDE_DIR "${CUDAToolkit_INCLUDE_DIRS}")
    set(CUTE_CUDA_LIBRARY_DIR "${CUDAToolkit_LIBRARY_DIR}")
    if(CUTE_DSL_CUDA_ROOT)
        find_path(CUTE_CUDA_HEADERS cuda_runtime.h PATHS "${CUTE_DSL_CUDA_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
        find_library(CUTE_CUDA_RUNTIME_LIBRARY NAMES cudart libcudart.so.13
            PATHS "${CUTE_DSL_CUDA_ROOT}/lib64" "${CUTE_DSL_CUDA_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
        get_filename_component(CUTE_CUDA_LIBRARY_DIR "${CUTE_CUDA_RUNTIME_LIBRARY}" DIRECTORY)
        add_library(CuteCUDARuntime SHARED IMPORTED)
        set_target_properties(CuteCUDARuntime PROPERTIES IMPORTED_LOCATION "${CUTE_CUDA_RUNTIME_LIBRARY}")
        set(CUTE_CUDA_TARGET CuteCUDARuntime)
        set(CUTE_CUDA_INCLUDE_DIR "${CUTE_CUDA_HEADERS}")
    endif()

    set(cute_objects)
    foreach(dtype f32 f16 bf16)
        require_artifact("${CUTE_DSL_ARTIFACT_DIR}/rmsnorm_${dtype}.h" test_cute_dsl_rmsnorm.py)
        require_artifact("${CUTE_DSL_ARTIFACT_DIR}/rmsnorm_${dtype}.o" test_cute_dsl_rmsnorm.py)
        list(APPEND cute_objects "${CUTE_DSL_ARTIFACT_DIR}/rmsnorm_${dtype}.o")
    endforeach()
    set_source_files_properties(${cute_objects} PROPERTIES EXTERNAL_OBJECT TRUE)
    add_library(cute_dsl_rmsnorm_kernels SHARED ${cute_objects})
    set_target_properties(cute_dsl_rmsnorm_kernels PROPERTIES LINKER_LANGUAGE CXX)
    target_link_libraries(cute_dsl_rmsnorm_kernels PRIVATE CuteDSLRuntime ${CUTE_CUDA_TARGET} CUDA::cuda_driver)

    add_rmsnorm_application(test_cute_dsl_rmsnorm)
    target_include_directories(test_cute_dsl_rmsnorm SYSTEM BEFORE PRIVATE
        "${CUTE_CUDA_INCLUDE_DIR}" "${CUTE_DSL_ARTIFACT_DIR}")
    target_link_libraries(test_cute_dsl_rmsnorm PRIVATE CuteDSLRuntime ${CUTE_CUDA_TARGET})
    add_dependencies(test_cute_dsl_rmsnorm cute_dsl_rmsnorm_kernels)
    if(BUILD_TESTING)
        add_test(NAME cute_dsl_rmsnorm_object COMMAND test_cute_dsl_rmsnorm "${CUTE_DSL_ARTIFACT_DIR}")
        add_test(NAME cute_dsl_rmsnorm_shared COMMAND test_cute_dsl_rmsnorm
            "${CUTE_DSL_ARTIFACT_DIR}" "$<TARGET_FILE:cute_dsl_rmsnorm_kernels>")
        configure_rmsnorm_test(cute_dsl_rmsnorm_object "${CUTE_CUDA_LIBRARY_DIR}" "${CUTE_DSL_LIBRARY_DIR}")
        configure_rmsnorm_test(cute_dsl_rmsnorm_shared "${CUTE_CUDA_LIBRARY_DIR}" "${CUTE_DSL_LIBRARY_DIR}")
    endif()
endif()

if(ENABLE_TILELANG)
    if(NOT TVM_FFI_ROOT OR NOT TILELANG_ROOT)
        message(FATAL_ERROR "Set TVM_FFI_ROOT and TILELANG_ROOT, or disable ENABLE_TILELANG")
    endif()
    find_path(TVM_FFI_INCLUDE_DIR tvm/ffi/function.h PATHS "${TVM_FFI_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
    find_library(TVM_FFI_LIBRARY tvm_ffi PATHS "${TVM_FFI_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    find_library(TILELANG_RUNTIME_LIBRARY tvm PATHS "${TILELANG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    get_filename_component(TVM_FFI_LIBRARY_DIR "${TVM_FFI_LIBRARY}" DIRECTORY)
    add_library(TVMFFI SHARED IMPORTED)
    set_target_properties(TVMFFI PROPERTIES
        IMPORTED_LOCATION "${TVM_FFI_LIBRARY}"
        INTERFACE_SYSTEM_INCLUDE_DIRECTORIES "${TVM_FFI_INCLUDE_DIR}"
        INTERFACE_INCLUDE_DIRECTORIES "${TVM_FFI_INCLUDE_DIR}")
    foreach(dtype f32 f16 bf16)
        foreach(width 33 128 4096)
            require_artifact("${TILELANG_ARTIFACT_DIR}/rmsnorm_${dtype}_h${width}.so" test_tilelang_rmsnorm.py)
        endforeach()
    endforeach()
    add_rmsnorm_application(test_tilelang_rmsnorm)
    target_link_libraries(test_tilelang_rmsnorm PRIVATE TVMFFI CUDA::cudart)
    if(BUILD_TESTING)
        add_test(NAME tilelang_rmsnorm COMMAND test_tilelang_rmsnorm
            "${TILELANG_ARTIFACT_DIR}" "${TILELANG_RUNTIME_LIBRARY}")
        configure_rmsnorm_test(tilelang_rmsnorm "${CUDAToolkit_LIBRARY_DIR}" "${TVM_FFI_LIBRARY_DIR}")
    endif()
endif()

if(ENABLE_TRITON)
    require_artifact("${TRITON_ARTIFACT_DIR}/rmsnorm_symbols.h" test_triton_rmsnorm.py)
    require_artifact("${TRITON_ARTIFACT_DIR}/rmsnorm_sources.cmake" test_triton_rmsnorm.py)
    include("${TRITON_ARTIFACT_DIR}/rmsnorm_sources.cmake")
    if(NOT TRITON_RMSNORM_SOURCES)
        message(FATAL_ERROR "The Triton export manifest contains no C launchers")
    endif()
    list(TRANSFORM TRITON_RMSNORM_SOURCES PREPEND "${TRITON_ARTIFACT_DIR}/")
    foreach(source IN LISTS TRITON_RMSNORM_SOURCES)
        require_artifact("${source}" test_triton_rmsnorm.py)
    endforeach()
    add_library(triton_rmsnorm_kernels SHARED ${TRITON_RMSNORM_SOURCES})
    target_compile_features(triton_rmsnorm_kernels PRIVATE c_std_11)
    set_target_properties(triton_rmsnorm_kernels PROPERTIES C_EXTENSIONS OFF)
    target_compile_options(triton_rmsnorm_kernels PRIVATE -Wall -Wextra)
    target_link_libraries(triton_rmsnorm_kernels PRIVATE CUDA::cuda_driver)

    add_rmsnorm_application(test_triton_rmsnorm)
    target_include_directories(test_triton_rmsnorm PRIVATE "${TRITON_ARTIFACT_DIR}")
    target_link_libraries(test_triton_rmsnorm PRIVATE CUDA::cudart)
    add_dependencies(test_triton_rmsnorm triton_rmsnorm_kernels)
    if(BUILD_TESTING)
        add_test(NAME triton_rmsnorm COMMAND test_triton_rmsnorm "$<TARGET_FILE:triton_rmsnorm_kernels>")
        configure_rmsnorm_test(triton_rmsnorm "${CUDAToolkit_LIBRARY_DIR}")
    endif()
endif()
