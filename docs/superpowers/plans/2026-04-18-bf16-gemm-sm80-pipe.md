**Goal:** Create `bf16_gemm_sm80_pipe.cu` — a pipelined BF16 GEMM kernel using cp.async 3-stage software pipeline, overlapping gmem→smem transfers with tensor core computation.

**Architecture:** Copy `bf16_gemm_sm80_opt.cu` (swizzle + LDSM) and replace synchronous gmem→smem copy with `SM80_CP_ASYNC_CACHEALWAYS<uint128_t>` via `make_tiled_copy`. Add 3-stage smem buffering with prefetch/main-loop/drain phases. Uses `cp_async_fence()` + `cp_async_wait<N>()` for async synchronization, inner `k_block` loop for fine-grained overlap. Mirrors the CUTLASS `sgemm_sm80.cu` tutorial pattern exactly.

**Tech Stack:** CUDA (SM80+), CuTe (CUTLASS 3.x), `cp.async` PTX instructions, `ldmatrix` (LDSM), XOR swizzle

```cpp
 * BF16 GEMM using SM80 tensor cores with CuTe — Pipelined Sample
 *
 * Demonstrates (on top of the optimized sample):
 *   - cp.async 3-stage software pipeline for gmem→smem overlap
 *   - make_tiled_copy with SM80_CP_ASYNC_CACHEALWAYS for vectorized async gmem→smem

 *   B: bf16, K x N, stored as (N, K) in CuTe with K-contiguous stride
 *   C: f32, M x N, column-major
 *
 * Target: SM80+ tensor cores (runs on SM90 L20Y GPU)
 *
 * Compared to bf16_gemm_sm80_opt.cu (swizzle + LDSM, synchronous):
 *   - gmem→smem: synchronous copy → cp.async 128-bit async transfers

 *   - Sync: 2× __syncthreads() → cp_async_fence/wait + __syncthreads()
 *   - Everything else unchanged (MMA atom, swizzle, LDSM, bM, bN, bK)
 *

 *   The 3-stage pipeline overlaps gmem→smem async copies with smem→regs loads
 *   and tensor core MMA computation. While MMA works on data from stage N,
 *   cp.async loads stage N+1 from global memory.
```

After the G2S partitioning (Step 3) and before the MMA setup (Step 4), insert the prefetch phase. Add this new section:

```cpp
  // ---- Step 3b: Prefetch (fill the pipeline) ----
  //
  // Issue bP-1 async gmem→smem copies to fill pipeline stages 0..bP-2.
  // Each copy is committed with cp_async_fence() so cp_async_wait can track it.
  //
  // k_tile_count tracks remaining tiles (decremented for each issued copy).
  // k_tile_next tracks which gmem K-tile to read next.
  //
  // After this loop, stages 0 and 1 are in-flight (or completed).
  // Stage 2 will be filled in the main loop.

  auto K_PIPE_MAX = size<3>(tAsA);                                       // = bP = 3

  int k_tile_count = size<3>(tAgA);                                      // total K-tiles
  int k_tile_next  = 0;                                                  // next gmem tile to read

  CUTE_UNROLL
  for (int k_pipe = 0; k_pipe < K_PIPE_MAX - 1; ++k_pipe) {
    copy(g2s_copy_a, tAgA(_,_,_,k_tile_next), tAsA(_,_,_,k_pipe));
    copy(g2s_copy_b, tBgB(_,_,_,k_tile_next), tBsB(_,_,_,k_pipe));
    cp_async_fence();
    --k_tile_count;
    if (k_tile_count > 0) { ++k_tile_next; }
  }
```

- [ ] **Step 12: Replace main loop with pipelined version**

Replace the entire Step 5 (main loop) section:

```cpp
  // ---- Step 5: Pipelined main loop ----
  //
  // Three-phase pipeline: prefetch (done above), main loop, drain.
  //
  // Pipe state:
  //   smem_pipe_read  — which smem stage to load from (S2R reads from here)
  //   smem_pipe_write — which smem stage to write to (G2S async writes here)
  //   k_tile_count    — remaining K-tiles (counts down to -(bP-1) to drain)
  //   k_tile_next     — next gmem K-tile index to read
  //
  // Inner k_block loop:
  //   The smem buffer (BLK_K=64) is processed in K_BLOCK_MAX MMA atom steps.
  //   K_BLOCK_MAX = size<2>(tCrA) = bK / MMA_K = 64 / 16 = 4.
  //
  //   At k_block == 0 (start of new K-tile):
  //     - Issue async G2S copy for the next K-tile
  //     - Advance pipe pointers
  //   At k_block == K_BLOCK_MAX-1 (end of K-tile):
  //     - Wait for next smem stage to be ready
  //     - Reslice the S2R source tensor
  //   Each k_block:
  //     - Load next k_block's smem→regs (LDSM)
  //     - MMA compute on current k_block's registers
  //
  // Overlap achieved:
  //   - cp.async (gmem→smem) runs in parallel with LDSM + MMA
  //   - LDSM (smem→regs) for k_block+1 runs in parallel with MMA for k_block
  //   - For large tiles, cp.async finishes before we need the data → no stall

  int smem_pipe_read  = 0;
  int smem_pipe_write = K_PIPE_MAX - 1;

  // Current smem slice for S2R reads
  Tensor tXsA_p = tXsA(_,_,_,smem_pipe_read);
  Tensor tXsB_p = tXsB(_,_,_,smem_pipe_read);

  // Number of K-blocks per smem stage (MMA atom iterations in K)
  auto K_BLOCK_MAX = size<2>(tCrA);                                      // = 4

  // Register prefetch: wait for the first smem stage, load first k_block
  if (K_BLOCK_MAX > 1) {
    cp_async_wait<K_PIPE_MAX - 2>();
    __syncthreads();
    copy(s2r_atom_a, tXsA_p(_,_,Int<0>{}), tXrA(_,_,Int<0>{}));
    copy(s2r_atom_b, tXsB_p(_,_,Int<0>{}), tXrB(_,_,Int<0>{}));
  }

  // Pipelined main loop
  // CUTE_NO_UNROLL prevents the compiler from unrolling the outer loop,
  // which would blow up register pressure. The inner k_block loop is unrolled.
  CUTE_NO_UNROLL
  while (k_tile_count > -(K_PIPE_MAX - 1))
  {
    CUTE_UNROLL
    for (int k_block = 0; k_block < K_BLOCK_MAX; ++k_block)
    {
      // At the last k_block: wait for next smem stage, update S2R slice
      if (k_block == K_BLOCK_MAX - 1)
      {
        tXsA_p = tXsA(_,_,_,smem_pipe_read);
        tXsB_p = tXsB(_,_,_,smem_pipe_read);

        // Wait until all but (bP-2) commit groups are done.
        // With bP=3, this waits for the oldest outstanding group.
        cp_async_wait<K_PIPE_MAX - 2>();
        __syncthreads();
      }

      // Load smem → registers for the NEXT k_block (LDSM)
      // This overlaps with the MMA compute for the CURRENT k_block below.
      auto k_block_next = (k_block + Int<1>{}) % K_BLOCK_MAX;
      copy(s2r_atom_a, tXsA_p(_,_,k_block_next), tXrA(_,_,k_block_next));
      copy(s2r_atom_b, tXsB_p(_,_,k_block_next), tXrB(_,_,k_block_next));

      // At the first k_block: issue new async G2S copy, advance pipe
      if (k_block == 0)
      {
        copy(g2s_copy_a, tAgA(_,_,_,k_tile_next), tAsA(_,_,_,smem_pipe_write));
        copy(g2s_copy_b, tBgB(_,_,_,k_tile_next), tBsB(_,_,_,smem_pipe_write));
        cp_async_fence();

        // Advance gmem tile counter
        --k_tile_count;
        if (k_tile_count > 0) { ++k_tile_next; }

        // Advance smem pipe pointers
        // write moves to the stage we just finished reading
        // read moves to the next stage (that was just filled by async)
        smem_pipe_write = smem_pipe_read;
        smem_pipe_read  = (smem_pipe_read == K_PIPE_MAX - 1) ? 0 : smem_pipe_read + 1;
      }

      // Tensor core MMA: tCrC += tCrA[k_block] * tCrB[k_block]
      gemm(mma, tCrA(_,_,k_block), tCrB(_,_,k_block), tCrC);
    }
  }
```

```cpp
void
bf16_gemm_tn(int m, int n, int k,
             Alpha alpha,
             bf16_t const* A, int ldA,
             bf16_t const* B, int ldB,
             Beta beta,
             float* C, int ldC,
             cudaStream_t stream = 0)
{
  using namespace cute;

  // Problem shape (dynamic)
  auto M = int(m);
  auto N = int(n);
  auto K = int(k);
  auto prob_shape = make_shape(M, N, K);                                // (M, N, K)

  // TN strides (mixed static/dynamic)
  auto dA = make_stride(ldA, Int<1>{});
  auto dB = make_stride(ldB, Int<1>{});
  auto dC = make_stride(Int<1>{}, ldC);

  // CTA tile sizes (static) — same as opt sample
  auto bM = Int<128>{};
  auto bN = Int<128>{};
  auto bK = Int<64>{};
  auto cta_tiler = make_shape(bM, bN, bK);                              // (128, 128, 64)

  // Pipeline stages (static)
  auto bP = Int<3>{};

  // Smem layouts (static, swizzled with PIPE dimension)
  //
  // Same Swizzle<3,3,3> atom as the opt sample, but tile_to_shape now
  // takes a 3D shape (bM, bK, bP). The pipe dimension bP=3 replicates
  // the swizzle pattern across 3 stages.
  auto swizzle_atom = composition(
      Swizzle<3, 3, 3>{},
      Layout<Shape <_8, Shape<_8, _8>>,
             Stride<_8, Stride<_1, _64>>>{});

  auto sA = tile_to_shape(swizzle_atom, make_shape(bM, bK, bP));        // (128, 64, 3) swizzled
  auto sB = tile_to_shape(swizzle_atom, make_shape(bN, bK, bP));        // (128, 64, 3) swizzled
  auto sC = make_layout(make_shape(bM, bN));                            // (128, 128) — unused

  // TiledMMA — same as opt sample
  TiledMMA mma = make_tiled_mma(
      SM80_16x8x16_F32BF16BF16F32_TN{},
      Layout<Shape<_2, _2>>{},
      Tile<_32, _32, _16>{});

  static_assert(decltype(size(mma))::value == 128, "Expected 128 threads");

  // G2S (global→shared) TiledCopy for async cp.async
  //
  // Copy atom: SM80_CP_ASYNC_CACHEALWAYS<uint128_t>
  //   Wraps cp.async.ca.shared.global.L2::128B — moves 128 bits per instruction.
  //   Each instruction transfers 8 contiguous bf16 elements (128 bits).
  //   CACHEALWAYS: data flows through L2 → L1 → shared memory.
  //
  // Thread layout: (16, 8) stride (8, 1)
  //   16 threads in M, 8 in K = 128 threads total, K-major order.
  //   Each thread covers (128/16, 64/8) = (8, 8) elements across iterations.
  //
  // Value layout: (1, 8)
  //   1 value in M, 8 in K per instruction = 8 bf16 = 128 bits.
  //   K-contiguous: matches our row-major A / K-contiguous B layouts.

  auto g2s_copy_a = make_tiled_copy(
      Copy_Atom<SM80_CP_ASYNC_CACHEALWAYS<uint128_t>, bf16_t>{},
      Layout<Shape<_16, _8>, Stride<_8, _1>>{},
      Layout<Shape<_1, _8>>{});

  auto g2s_copy_b = make_tiled_copy(
      Copy_Atom<SM80_CP_ASYNC_CACHEALWAYS<uint128_t>, bf16_t>{},
      Layout<Shape<_16, _8>, Stride<_8, _1>>{},
      Layout<Shape<_1, _8>>{});

  // S2R (smem→register) copy atoms — same as opt sample
  Copy_Atom<SM75_U32x4_LDSM_N, bf16_t> s2r_atom_a;
  Copy_Atom<SM75_U32x4_LDSM_N, bf16_t> s2r_atom_b;

  // Smem size calculation
  // SharedStorage uses ArrayEngine with cosize_v of the full 3D layouts.
  // This accounts for swizzle padding across all 3 stages.
  int smem_size = int(sizeof(SharedStorage<bf16_t, bf16_t, decltype(sA), decltype(sB)>));

  // Grid and block dimensions
  dim3 dimBlock(size(mma));
  dim3 dimGrid(size(ceil_div(M, bM)), size(ceil_div(N, bN)));

  // Get kernel function pointer for cudaFuncSetAttribute
  auto kernel_fptr = bf16_gemm_device<
      decltype(prob_shape), decltype(cta_tiler),
      bf16_t, decltype(dA), decltype(sA), decltype(g2s_copy_a), decltype(s2r_atom_a),
      bf16_t, decltype(dB), decltype(sB), decltype(g2s_copy_b), decltype(s2r_atom_b),
      float, decltype(dC), decltype(sC), decltype(mma),
      Alpha, Beta>;

  // Request enough shared memory for the 3-stage pipeline (~96 KB)
  cudaFuncSetAttribute(
      kernel_fptr,
      cudaFuncAttributeMaxDynamicSharedMemorySize, smem_size);

  // Prefer maximum shared memory carveout (L1 vs shared)
  cudaFuncSetAttribute(
      kernel_fptr,
      cudaFuncAttributePreferredSharedMemoryCarveout, 100);

  // Launch kernel with dynamic smem size
  kernel_fptr<<<dimGrid, dimBlock, smem_size, stream>>>(
      prob_shape, cta_tiler,
      A, dA, sA, g2s_copy_a, s2r_atom_a,
      B, dB, sB, g2s_copy_b, s2r_atom_b,
      C, dC, sC, mma,
      alpha, beta);
}
```

```cpp
int main(int argc, char** argv)
{
  using namespace cute;

  printf("BF16 GEMM (SM80 tensor cores, cp.async 3-stage pipeline)\n\n");

  float alpha = 1.0f;
  float beta  = 0.0f;

  // ---- Verify correctness once at 1024^3 ----
  {
    int m = 1024, n = 1024, k = 1024;
    int ldA = k, ldB = k, ldC = m;

    thrust::host_vector<bf16_t> h_A(m * k), h_B(n * k);
    thrust::host_vector<float>  h_C(m * n);
    for (int i = 0; i < m * k; ++i) h_A[i] = static_cast<bf16_t>(2.0 * (rand() / double(RAND_MAX)) - 1.0);
    for (int i = 0; i < n * k; ++i) h_B[i] = static_cast<bf16_t>(2.0 * (rand() / double(RAND_MAX)) - 1.0);
    for (int i = 0; i < m * n; ++i) h_C[i] = -1.0f;

    thrust::device_vector<bf16_t> d_A = h_A, d_B = h_B;
    thrust::device_vector<float>  d_C = h_C;

    bf16_gemm_tn(m, n, k, alpha,
                 d_A.data().get(), ldA,
                 d_B.data().get(), ldB,
                 beta,
                 d_C.data().get(), ldC);
    CUTE_CHECK_LAST();

    thrust::host_vector<float> h_result = d_C;

    // CPU reference: C[m,n] = alpha * sum_k A[m,k] * B[n,k] + beta * C[m,n]
    thrust::host_vector<float> h_ref(m * n, 0.0f);
    for (int i = 0; i < m; ++i)
      for (int j = 0; j < n; ++j) {
        float sum = 0.0f;
        for (int l = 0; l < k; ++l)
          sum += float(h_A[i * k + l]) * float(h_B[j * k + l]);
        h_ref[i + j * ldC] = alpha * sum + beta * h_C[i + j * ldC];
      }

    float max_err = 0.0f;
    for (int i = 0; i < m * n; ++i)
      max_err = std::max(max_err, std::abs(h_result[i] - h_ref[i]));

    printf("Correctness (1024^3): max error %e — %s\n\n", max_err, max_err < 0.01f ? "PASS" : "FAIL");
    if (max_err >= 0.01f) return 1;
  }

  // ---- Benchmark ----
  printf("Benchmark (100 iterations each):\n");
  benchmark_size(512,  512,  512,  alpha, beta, 0);
  benchmark_size(1024, 1024, 1024, alpha, beta, 0);
  benchmark_size(2048, 2048, 2048, alpha, beta, 0);
  benchmark_size(4096, 4096, 4096, alpha, beta, 0);

  return 0;
}
```
